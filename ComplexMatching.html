-- 2-dimensional cubical complexes
--
-- A square sits on four vertices  a — b
--                                 |   |
--                                 c — d
-- with top a→b, bottom c→d, left a→c, right b→d.
--
-- This is the cubical library's convention for `Square`, one dimension down:
-- its corners a₀₀ a₀₁ a₁₀ a₁₁ are our a b c d, and it takes its sides in the
-- order `Square a₀₋ a₁₋ a₋₀ a₋₁` = top, bottom, left, right, which is the
-- order `Sq` takes its edges.  So a realised 2-cell is a `Square` on the
-- realised boundary edges, with nothing reshuffled — see `CType2`.

-- The bundled cells and the boundary map depend only on the *raw* data
-- (V , E , Sq), not on any of the properties a Complex asserts.  They live in
-- their own module so that `Complex` can state `∂-inj` — which mentions ∂ — as
-- an honest field: declarations sitting between two fields of a record are
-- elaborated as let-bindings, so they may not pattern match (∂ splits on Side),
-- may not use `with`, and may not share a signature (`src tgt : …`).
module Cells (V : Type ℓ) (E : V → V → Type ℓ)
    (Sq : {a b c d : V} → E a b → E c d → E a c → E b d → Type ℓ) where

  -- the 1-cells are the graph development's, so that `Graph`'s `Edges-discrete`
  -- and `Edges-isHSet` apply to them verbatim
  open GM.Edge E public

  -- bundled 2-cells
  SquareOn : (a b c d : V) → Type ℓ
  SquareOn a b c d =
    Σ[ t ∈ E a b ] Σ[ bo ∈ E c d ] Σ[ l ∈ E a c ] Σ[ r ∈ E b d ] Sq t bo l r

  Squares : Type ℓ
  Squares = Σ[ a ∈ V ] Σ[ b ∈ V ] Σ[ c ∈ V ] Σ[ d ∈ V ] SquareOn a b c d

  -- boundary
  ∂ : Squares → Side → Edges
  ∂ (a , b , c , d , t , bo , l , r , _) tp = a , b , t
  ∂ (a , b , c , d , t , bo , l , r , _) bt = c , d , bo
  ∂ (a , b , c , d , t , bo , l , r , _) lf = a , c , l
  ∂ (a , b , c , d , t , bo , l , r , _) rt = b , d , r

record Complex (V : Type ℓ) (E : V → V → Type ℓ)
    (Sq : {a b c d : V} → E a b → E c d → E a c → E b d → Type ℓ)
    : Type ℓ where
  open Cells V E Sq public

  field
    graph : Graph V E

    Sq-isHProp : {a b c d : V} (t : E a b) (bo : E c d) (l : E a c) (r : E b d)
      → isHProp (Sq t bo l r)

    -- non-degeneracy: the four boundary edges of a square are pairwise
    -- distinct.  This makes "being a face" a proposition.
    ∂-inj : (s : Squares) {i j : Side} → ∂ s i ≡ ∂ s j → i ≡ j

  open Graph graph public

  -- properties of total space of edge and squares inherited from 

  SquareOn-isHProp : (a b c d : V) → isHProp (SquareOn a b c d)
  SquareOn-isHProp a b c d =
    isPropΣ (E-isHProp _ _) λ t → isPropΣ (E-isHProp _ _) λ bo →
    isPropΣ (E-isHProp _ _) λ l → isPropΣ (E-isHProp _ _) λ r → Sq-isHProp t bo l r


  Squares-discrete : Discrete Squares
  Squares-discrete (a , b , c , d , s) (a' , b' , c' , d' , s')
    with V-discrete a a' | V-discrete b b' | V-discrete c c' | V-discrete d d'
  ... | no ¬p | _      | _      | _      = no λ r → ¬p (cong fst r)
  ... | yes p | no ¬q  | _      | _      = no λ r → ¬q (cong (fst ∘ snd) r)
  ... | yes p | yes q  | no ¬o  | _      = no λ r → ¬o (cong (fst ∘ snd ∘ snd) r)
  ... | yes p | yes q  | yes o  | no ¬n  = no λ r → ¬n (cong (fst ∘ snd ∘ snd ∘ snd) r)
  ... | yes p | yes q  | yes o  | yes n  = yes (ΣPathP (p , ΣPathP (q , ΣPathP (o , ΣPathP (n ,
        isProp→PathP (λ i → SquareOn-isHProp (p i) (q i) (o i) (n i)) s s')))))

  Squares-isHSet : isHSet Squares
  Squares-isHSet = Discrete→isSet Squares-discrete

-- Matchings

module _ {V : Type ℓ} {E : V → V → Type ℓ}
    {Sq : {a b c d : V} → E a b → E c d → E a c → E b d → Type ℓ}
    (K : Complex V E Sq) where

  open Complex K

  -- The raw data of a full matching:
  --   · μ̂₁ pairs vertices with edges, exactly as in the graph case;
  --   · μ₁ pairs every square with one of its own faces.  Because μ₁ lands in
  --     Side, fullness ("every square is matched") and "each square is in
  --     exactly one pair, with one of its faces" hold by construction.
  record MatchingData : Type ℓ where
    field
      μ₀ : V → V
      μ₁ : Squares → Side

    -- the edge a square is matched with
    μ̂₁ : Squares → Edges
    μ̂₁ s = ∂ s (μ₁ s)

    -- μ̂₁̂◃ : (s : Squares) → μ̂₁̂ s ◃ s
    -- μ̂₁̂◃ s = μ₁ s , refl

    -- an edge can be matched upwards (with a square) or downwards (with its
    -- source vertex)
    -- (separate signatures: a shared one would form a mutual block, which is
    -- not allowed before the record's last field)
    isUp : Edges → Type ℓ
    isUp e = Σ[ s ∈ Squares ] μ̂₁ s ≡ e

    isDn : Edges → Type ℓ
    isDn e = let (u , v , _) = e in μ₀ u ≡ v

    field
      -- each edge caps at most one square …
      μ̂₁-inj : {s s' : Squares} → μ̂₁ s ≡ μ̂₁ s' → s ≡ s'
      -- … and is never matched both upwards and downwards.  Together with μ̂₁
      -- and μ₁ being functions, this says intrinsically that every cell lies
      -- in at most one matched pair.
      up∩dn : (e : Edges) → isUp e → ¬ isDn e
      -- Unlike being matched downwards, being matched upwards is a search over
      -- squares rather than a consequence of `V-discrete`, so its decidability
      -- has to be asserted.  It is part of the matching datum because
      -- everything the matching is *for* — the collapse of `ComplexMorse`, the
      -- decision procedures below — needs it, and because whoever writes down
      -- a matching is exactly who can supply it (it holds e.g. as soon as the
      -- complex is finite).
      isUp? : (e : Edges) → Dec (isUp e)

  -- Gradient-path machinery over a fixed matching datum.
  module GradientPaths (m : MatchingData) where
    open MatchingData m public

    ----------------------------------------------------------------------
    -- dimension 0: literally the graph development
    open GM.Paths graph μ₀ public -- using (Crit₀ ; _⇝_ ;  _ᵥ ; acyclic ; mp)
    -- (this exports isCrit, Crit, _ᵥ, _⇒ᵐ_, _⇒ᶜ_, _⇝_, acyclic, Crit₁, …)

    ----------------------------------------------------------------------
    -- propositionality of the matching relations on edges

    isUp-isHProp : (e : Edges) → isHProp (isUp e)
    isUp-isHProp e (s , p) (s' , q) =
      Σ≡Prop (λ _ → Edges-isHSet _ _) (μ̂₁-inj (p ∙ sym q))

    isDn-isHProp : (e : Edges) → isHProp (isDn e)
    isDn-isHProp e = V-isHSet _ _

    ----------------------------------------------------------------------
    -- Critical cells
    --
    -- Critical vertices are as in the graph case (isCrit₀ / Crit₀ above).
    -- A Critical edge is matched neither up nor down.  There are *no*
    -- Critical squares: fullness is built into μ₁.

    isCrit₁ : Edges → Type ℓ
    isCrit₁ e = (¬ isDn e) × (¬ isUp e)

    isCrit₁-isHProp : (e : Edges) → isHProp (isCrit₁ e)
    isCrit₁-isHProp e = isProp× (isProp¬ _) (isProp¬ _)

    Crits₁ : Type ℓ
    Crits₁ = Σ[ e ∈ Edges ] isCrit₁ e

    Crits₁-isHSet : isHSet Crits₁
    Crits₁-isHSet = isSetΣSndProp Edges-isHSet isCrit₁-isHProp

    ----------------------------------------------------------------------
    -- gradient steps and paths in dimension one
    --
    -- A step leaves an edge e through the square matched with it, exiting
    -- along a *different* face.  Note the branching: a square has three exit
    -- faces, so unlike dimension zero these steps are not deterministic.

    -- exitData : Squares → Edges → Type ℓ
    -- exitData s f = Σ[ i ∈ Side ] (¬ i ≡ μ₁ s) × (∂ s i ≡ f)

    -- exitData-isHProp : (s : Squares) (f : Edges) → isHProp (exitData s f)
    -- exitData-isHProp s f (i , _ , p) (j , _ , q) =
    --   Σ≡Prop (λ _ → isProp× (isProp¬ _) (Edges-isHSet _ _)) (∂-inj s (p ∙ sym q))

    _⇒₁_ : Edges → Edges → Type ℓ
    e ⇒₁ f = Σ[ s ∈ Squares ] (μ̂₁ s ≡ e) × (Σ[ i ∈ Side ] (¬ i ≡ μ₁ s) × (∂ s i ≡ f)) -- exitData s f

    -- for a *fixed* target the step is a proposition (the square is forced
    -- by μ̂₁̂-inj, the exit side by ∂-inj)
    -- ⇒₁-isHProp : {e f : Edges} → isHProp (e ⇒₁ f)
    -- ⇒₁-isHProp {e} {f} (s , p , x) (s' , p' , x') =
    --   Σ≡Prop (λ t → isProp× (Edges-isHSet _ _) (exitData-isHProp t f))
    --     (μ̂₁-inj (p ∙ sym p'))

    upStep : {e f : Edges} → e ⇒₁ f → isUp e
    upStep (s , p , _) = s , p

    -- paths terminate at edges that are not matched upwards (i.e. Critical
    -- edges, or edges matched down to a vertex) — the exact analogue of
    -- `end` requiring isCrit₀ in dimension zero.
    data _⇝₁_ : Edges → Edges → Type ℓ where
      end : {e f : Edges} → ¬ isUp e → e ≡ f → e ⇝₁ f
      step  : ∀ {e} (f) {g} → e ⇒₁ f → f ⇝₁ g → e ⇝₁ g

    -- endNoStep : {e f : Edges} → ¬ isUp e → ¬ (e ⇒₁ f)
    -- endNoStep ¬up stp = ¬up (upStep stp)

    -- _∙₁_ : {e f g : Edges} → e ⇝₁ f → f ⇝₁ g → e ⇝₁ g
    -- _∙₁_ {g = g} (end ¬up p) q = subst (_⇝₁ g) (sym p) q
    -- _∙₁_ (step f x p) q = step f x (p ∙₁ q)

    ----------------------------------------------------------------------
    -- acyclicity in dimension one
    --
    -- Because steps branch, "every edge flows to a unique sink" is no longer
    -- the right notion (sinks need not be unique, and _⇝₁_ into a fixed
    -- target need not be a proposition).  The intrinsic formulation is that
    -- the step relation is well-founded: no infinite (in particular no
    -- cyclic) gradient path.
    --
    -- This works here because `end` asks only for ¬ isUp e — being unable to
    -- step *is* what it means to be done, so an accessibility witness already
    -- builds a gradient path (that is `sink₁` below).  In dimension zero
    -- `end` asks for isCrit₀ u, which is strictly stronger than being unable
    -- to step, so the same reading does not apply there; see `acyclic`.

    _<₁_ : Edges → Edges → Type ℓ
    f <₁ e = e ⇒₁ f

    acyclic₁ : Type ℓ
    acyclic₁ = WellFounded _<₁_

    ----------------------------------------------------------------------
    -- Morse boundary data
    --
    -- With a full matching there are no Critical squares, so the Morse
    -- complex is a graph on (Crit₀ , Crit₁).  The dimension-(0,1) boundary of
    -- a Critical edge is given by flowing its endpoints down to Critical
    -- vertices (this refines the inherited Crit₁, which ranged over all
    -- unmatched edges).

    -- ∂ᴹ : Crits₁ → Crit₀ → Type ℓ
    -- ∂ᴹ ((u , v , e) , _) c = (u ⇝ c ᵥ) ⊎ (v ⇝ c ᵥ)

  -- A (full, acyclic) Morse matching: matching data, acyclic in both
  -- dimensions.  Dimension zero keeps the graph formulation, which asserts
  -- more than well-foundedness of its steps: it also says that μ̂₁ pairs every
  -- non-Critical vertex with an *incident* one.  (Well-foundedness alone is
  -- vacuous on a graph with no edges, where every non-identity μ̂₁ leaves
  -- vertices that are neither Critical nor able to step.)
  2FullMatching : Type ℓ
  2FullMatching = Σ[ m ∈ MatchingData ]
      GradientPaths.acyclic m × WellFounded (GradientPaths._<₁_ m)

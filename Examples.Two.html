two₁ : Bool → Bool → Type
two₁ false false = ⊥
two₁ false true = ⊤
two₁ true false = ⊤
two₁ true true = ⊥

two : Graph Bool two₁
Graph.V-discrete two = _≟_
Graph.E-isIrrefl two false ()
Graph.E-isIrrefl two true ()
Graph.E-isHProp two false true = isHProp⊤
Graph.E-isHProp two true false = isHProp⊤

two₁? : (u v : Bool) → Dec (two₁ u v)
two₁? false false = no (λ a → a)
two₁? false true = yes tt
two₁? true false = yes tt
two₁? true true = no (λ a → a)

open Graph two

two-μ : Bool → Bool
two-μ true = true
two-μ false = true


module _ where
  open Paths two two-μ
  two-μ-acy : acyclic
  two-μ-acy false = (true , refl) , step {false , true , tt} refl (end refl refl)
  two-μ-acy true = (true , refl) , (end refl refl)

twoM : Matching two
twoM = (two-μ , two-μ-acy)

open Morse two twoM

⊤₁≡Mtwo : ∣ ⊤ , ⊤₁ ∣ ≡ ∣ Crit₀ , Crit₁ ∣
⊤₁≡Mtwo = isoReal F F₁ G G₁ S S₁ R R₁
  where
  F : ⊤ → Crit₀
  F tt = true , refl
  F₁ : (x y : ⊤) → ⊤₁ x y → Crit₁ (F x) (F y)
  F₁ tt tt tt = (true , false , tt) , true≢false , snd (two-μ-acy true) , snd (two-μ-acy false)
  G : Crit₀ → ⊤
  G _ = tt
  G₁ : (x y : Crit₀) → Crit₁ x y → ⊤₁ tt tt
  G₁ u v f = tt
  S : (x : Crit₀) → F (G x) ≡ x
  S (true , p) = ΣPathP (refl , (V-isHSet _ _ refl p))
  S (false , p) = absurd (true≢false p)
  S₁ : (x y : Crit₀) (f : Crit₁ x y) →
    PathP (λ i → Crit₁ (S x i) (S y i)) (F₁ (G x) (G y) (G₁ x y f)) f
  S₁ (false , b) (false , d) f = absurd (true≢false b)
  S₁ (false , b) (true , d) f = absurd (true≢false b)
  S₁ (true , b) (false , d) f = absurd (true≢false d)
  S₁ (true , b) (true , d) ((false , true , tt) , ¬μ , s , a) = absurd (¬μ refl)
  S₁ (true , b) (true , d) ((true , false , tt) , ¬μ , r , s) i = (true , false , tt) ,
    ⇒ᶜIsHProp {true} {false} true≢false ¬μ i ,
    ⇝IsHProp (true , refl) (end refl refl) r i ,
    ⇝IsHProp (true , refl) (snd (two-μ-acy false)) s i
  R : (x : ⊤) → G (F x) ≡ x
  R tt = refl
  R₁ : (x y : ⊤) (f : ⊤₁ x y) →
    PathP (λ i → ⊤₁ (R x i) (R y i)) (G₁ (F x) (F y) (F₁ x y f)) f
  R₁ tt tt tt = refl

∣two₁∣-Groupoid : isGroupoid ∣ Bool , two₁ ∣
∣two₁∣-Groupoid = subst isGroupoid
  (S¹≡∣⊤₁∣ ∙∙ ⊤₁≡Mtwo ∙∙ sym MorseTheorem) isGroupoidS¹


H¹-two≅ℤ : GroupIso (coHomGr 1 (∣ Bool , two₁ ∣)) ℤGroup
H¹-two≅ℤ = subst (λ A → GroupIso (coHomGr 1 A) ℤGroup)
  (S¹≡∣⊤₁∣ ∙∙ ⊤₁≡Mtwo ∙∙ sym MorseTheorem) H¹-S¹≅ℤ

Hⁿ⁺²-two≅0 : (n : ℕ) → coHomGr (2 + n) ∣ Bool , two₁ ∣ ≅ ⊤Group
Hⁿ⁺²-two≅0 n = subst (λ A → GroupIso (coHomGr (2 + n) A) ⊤Group)
  (S¹≡∣⊤₁∣ ∙∙ ⊤₁≡Mtwo ∙∙ sym MorseTheorem) (Hⁿ-S¹≅0 n)

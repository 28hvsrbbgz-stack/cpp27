CType₂ : Type (ℓ-suc ℓ)
CType₂ {ℓ} = Σ[ A ∈ Type ℓ ] Σ[ B ∈ (A → A → Type ℓ) ] (∀{a b c d} → B a b → B c d → B a c → B b d → Type ℓ)


-- The realisation of a 2-dimensional complex: vertices become points, edges
-- become paths, and a 2-cell  Sq t bo l r  becomes a filler of the square
--
--        ∣ t ∣₁
--    a ---------> b            spelled with the cubical library's `Square`,
--    |            |            whose argument order  a₀₋ a₁₋ a₋₀ a₋₁  is
--  ∣ l ∣₁      ∣ r ∣₁          exactly the order  top bottom left right  in
--    |            |            which `Sq` takes its boundary edges.
--    v  ∣ bo ∣₁   v
--    c ---------> d
--
-- The four corners are implicit: they are determined by the boundary edges.
-- data ∣_,_,_∣ (V : Type ℓ) (E : V → V → Type ℓ)
--   (Sq : ∀{a b c d} → E a b → E c d → E a c → E b d → Type ℓ)
--   : Type ℓ where
--     ∣_∣₀ :  V → ∣ V , E , Sq ∣
--     ∣_⇒_∋_∣₁ : (x y : V) → E x y → ∣ x ∣₀ ≡ ∣ y ∣₀
--     ∣_,_,_,_∋_∣₂ : ∀ {a b c d}
--       (t : E a b) (bo : E c d) (l : E a c) (r : E b d) → Sq t bo l r
--       → Square ∣ a ⇒ b ∋ t ∣₁ ∣ c ⇒ d ∋ bo ∣₁ ∣ a ⇒ c ∋ l ∣₁ ∣ b ⇒ d ∋ r ∣₁
data ∣_∣ (C : CType₂ {ℓ}) : Type ℓ where
    ∣_∣₀ :  fst C → ∣ C ∣
    ∣_⇒_∋_∣₁ : (x y : fst C) → fst (snd C) x y → ∣ x ∣₀ ≡ ∣ y ∣₀
    ∣_,_,_,_∋_∣₂ : ∀ {a b c d}
      (t : fst (snd C) a b) (bo : fst (snd C) c d)
      (l : fst (snd C) a c) (r : fst (snd C) b d) → snd (snd C) t bo l r
      → PathP (λ i → ∣ a ⇒ c ∋ l ∣₁ i ≡ ∣ b ⇒ d ∋ r ∣₁ i) ∣ a ⇒ b ∋ t ∣₁ ∣ c ⇒ d ∋ bo ∣₁


infix 40 ∣_∣₀
infix 40 ∣_⇒_∋_∣₁
infix 40 ∣_,_,_,_∋_∣₂



import CType1 as GR


noSq : {V : Type ℓ} {E : V → V → Type ℓ} → ∣ V , E , (λ _ _ _ _ → ⊥*) ∣ ≡ GR.∣ V , E ∣
noSq {V = V} {E = E} = isoToPath (iso F G S R) where
  F : ∣ V , E , (λ _ _ _ _ → ⊥*) ∣ → GR.∣ V , E ∣
  F ∣ u ∣₀ = GR.∣ u ∣₀
  F (∣ u ⇒ v ∋ f ∣₁ i) = GR.∣ u ⇒ v ∋ f ∣₁ i
  F (∣ _ , _ , _ , _ ∋ lift () ∣₂ i j)
  G : GR.∣ V , E ∣ → ∣ V , E , (λ _ _ _ _ → ⊥*) ∣
  G GR.∣ u ∣₀ = ∣ u ∣₀
  G (GR.∣ u ⇒ v ∋ f ∣₁ i) = ∣ u ⇒ v ∋ f ∣₁ i
  S : (x : GR.∣ V , E ∣) → F (G x) ≡ x
  S GR.∣ u ∣₀ = refl
  S (GR.∣ u ⇒ v ∋ f ∣₁ i) = refl
  R : (y : ∣ V , E , (λ _ _ _ _ → ⊥*) ∣) → G (F y) ≡ y
  R ∣ u ∣₀ = refl
  R (∣ u ⇒ v ∋ f ∣₁ i) = refl
  R (∣ _ , _ , _ , _ ∋ lift () ∣₂ i j)

-- Maps out of the realisation
--
-- A morphism of complexes — a map on vertices, on edges over it, and on
-- squares over that — induces a map of realisations.  This is the `𝓕` half of
-- the `isoReal` pattern in `GraphRealisation`, one dimension up.

-- module _ {V : Type ℓ} {E : V → V → Type ℓ}
--          {Sq : {a b c d : V} → E a b → E c d → E a c → E b d → Type ℓ}
--          {V' : Type ℓ} {E' : V' → V' → Type ℓ}
--          {Sq' : {a b c d : V'} → E' a b → E' c d → E' a c → E' b d → Type ℓ}
--          (F : V → V') (F₁ : (x y : V) → E x y → E' (F x) (F y))
--          (F₂ : {a b c d : V} (t : E a b) (bo : E c d) (l : E a c) (r : E b d)
--              → Sq t bo l r
--              → Sq' (F₁ a b t) (F₁ c d bo) (F₁ a c l) (F₁ b d r))
--          where

--   𝓕 : ∣ V , E , Sq ∣ → ∣ V' , E' , Sq' ∣
--   𝓕 ∣ x ∣₀ = ∣ F x ∣₀
--   𝓕 (∣ u ⇒ v ∋ f ∣₁ i) = ∣ F u ⇒ F v ∋ F₁ u v f ∣₁ i
--   𝓕 (∣_,_,_,_∋_∣₂ {a} {b} {c} {d} t bo l r s i j) =
--     ∣ F₁ a b t , F₁ c d bo , F₁ a c l , F₁ b d r ∋ F₂ t bo l r s ∣₂ i j

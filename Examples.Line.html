module Examples.Line where

open import Cubical.Foundations.Prelude
open import Cubical.Data.Sigma
open import Cubical.Data.Nat
open import Cubical.Data.Nat.Order
open import Cubical.Data.Unit
open import Cubical.Data.Empty renaming (rec to absurd)
open import Cubical.Relation.Nullary

open import Prelude
open import GraphMatching
open import CType1
open import GraphMorse

line₁ : ℕ → ℕ → Type
line₁ zero n = ⊥
line₁ (suc m) n = m ≡ n

line : Graph ℕ line₁
Graph.V-discrete line = discreteℕ
Graph.E-isIrrefl line zero a = a
Graph.E-isIrrefl line (suc m) a = ¬m<m (0 , sym a)
Graph.E-isHProp line zero n = isHProp⊥
Graph.E-isHProp line (suc m) n = isSetℕ m n

line₁? : (u v : ℕ) → Dec (line₁ u v)
line₁? zero n = no (λ p → p)
line₁? (suc m) n = discreteℕ m n


open Graph line

line-μ : ℕ → ℕ
line-μ zero = zero
line-μ (suc n) = n

module _ where
  open Paths line line-μ
  line-acy : acyclic
  line-acy zero = (zero , refl) , (end refl refl)
  line-acy (suc m) = let IH = line-acy m in fst IH , step {suc m , m , refl} refl (snd IH)

lineM : Matching line
lineM = (line-μ , line-acy)


open Morse line lineM

⊥₁≡Mline : ∣ ⊤ , ⊥₁ ∣ ≡ ∣ Crit₀ , Crit₁ ∣
⊥₁≡Mline = isoReal F F₁ G G₁ S S₁ R R₁
  where
  F : ⊤ → Crit₀
  F tt = zero , refl
  F₁ : (x y : ⊤) → ⊥₁ x y → Crit₁ (F x) (F y)
  F₁ tt tt ()
  G : Crit₀ → ⊤
  G _ = tt
  G₁ : (x y : Crit₀) → Crit₁ x y → ⊥₁ tt tt
  G₁ (zero , p) (zero , q) ((suc m , n , e) , ¬μ , _ , _) = absurd (¬μ e)
  G₁ (zero , p) (suc n , q) = absurd (¬m<m (0 , sym q))
  G₁ (suc m , p) (zero , q) = absurd (¬m<m (0 , sym p))
  G₁ (suc m , p) (suc n , q) = absurd (¬m<m (0 , sym p))
  S : (x : Crit₀) → F (G x) ≡ x
  S (zero , p) = ΣPathP (refl , V-isHSet _ _ _ _)
  S (suc m , p) = absurd (¬m<m (0 , sym p))
  S₁ : (x y : Crit₀) (f : Crit₁ x y) →
    PathP (λ i → Crit₁ (S x i) (S y i)) (F₁ (G x) (G y) (G₁ x y f)) f
  S₁ (zero , p) (zero , q) ((suc m , n , e) , ¬μ , _ , _) = absurd (¬μ e)
  S₁ (zero , p) (suc n , q) = absurd (¬m<m (0 , sym q))
  S₁ (suc m , p) (zero , q) = absurd (¬m<m (0 , sym p))
  S₁ (suc m , p) (suc n , q) = absurd (¬m<m (0 , sym p))
  R : (x : ⊤) → G (F x) ≡ x
  R tt = refl
  R₁ : (x y : ⊤) (f : ⊥₁ x y) →
    PathP (λ i → ⊥₁ (R x i) (R y i)) (G₁ (F x) (F y) (F₁ x y f)) f
  R₁ tt tt ()


line-isContr : isContr ∣ ℕ , line₁ ∣
line-isContr = subst isContr (⊤≡∣⊥₁∣ ∙∙ ⊥₁≡Mline ∙∙ sym MorseTheorem) isContrUnit

{-# OPTIONS --cubical --safe #-}

-- The realisation of a graph with a spanning tree is a bouquet of circles, one
-- for every edge the tree leaves out — `Bouquet` in the sense of
-- `Cubical.HITs.Bouquet`, indexed by those edges.
--
-- This is `GraphForest` with the cycles put back in.  There the whole graph was
-- a forest and its realisation collapsed onto the set of roots; here the edges
-- come split into a tree part `T` and a rest `L`, and only `T` is contracted —
-- what survives is one loop for every edge of `L`.
--
-- The data asked for is the same as there, for the tree alone: a root, a path
-- running from it to every vertex, and — for every tree edge — the triangle
-- those paths span.  Two things make the extra edges cost nothing:
--
--   · the paths are asked for in the realisation of the *tree* `∣ V , T ∣`, so the
--     collapsing map kills them by construction (`trivial` below).  Asked for
--     in `∣ V , E ∣` they could run around a cycle, and the map built from them
--     would then conjugate the loops instead of preserving them;
--
--   · an edge of `L` from u to v is sent to the loop that runs from the root
--     down to u, along the edge, and back up from v — and `pqr⁻¹r≡pq` fills
--     the square saying that this is the edge again once the tree is gone.
module GraphBouquet where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.GroupoidLaws using (cong-∙∙)
open import Cubical.Foundations.Isomorphism
open import Cubical.Data.Sigma
open import Cubical.Data.Sum
open import Cubical.HITs.Bouquet.Base

open import Prelude using (pqr⁻¹r≡pq)
open import CType1

private
  variable
    ℓ : Level

module _ {V : Type ℓ} (T L : V → V → Type ℓ) where

  -- the edges are the disjoint union of tree and the remaining (loop) edges
  E : V → V → Type ℓ
  E u v = T u v ⊎ L u v

  -- we have one circle for every edge outside the tree
  Loops : Type ℓ
  Loops = Σ[ u ∈ V ] Σ[ v ∈ V ] L u v

  -- the tree sits inside the graph
  ι : ∣ V , T ∣ → ∣ V , E ∣
  ι ∣ v ∣₀ = ∣ v ∣₀
  ι (∣ u ⇒ v ∋ t ∣₁ i) = ∣ u ⇒ v ∋ inl t ∣₁ i

  CycleCollapse :
    (r : V)                                          -- the root
    (rootPath : (v : V) → Path (∣ V , T ∣) (∣ r ∣₀) (∣ v ∣₀))          -- a path down the tree
    (Fill : {u v : V} (t : T u v)                            -- and, for a tree edge the triangle it spans
      → PathP (λ i → Path (∣ V , T ∣) (∣ r ∣₀) (∣ u ⇒ v ∋ t ∣₁ i)) (rootPath u) (rootPath v))
    →  ∣ V , E ∣ ≡ Bouquet Loops
  CycleCollapse r rootPath Fill = isoToPath (iso to from sec ret) where
    -- the same path, read in the graph
    p : (v : V) → Path (∣ V , E ∣) (∣ r ∣₀) (∣ v ∣₀)
    p v = cong ι (rootPath v)

    -- Contracting the tree.  A vertex goes to the point of the bouquet, a
    -- tree edge stays there, and an edge of `L` becomes its own circle.
    to : ∣ V , E ∣ → Bouquet Loops
    to ∣ v ∣₀ = base
    to (∣ u ⇒ v ∋ inl t ∣₁ i) = base
    to (∣ u ⇒ v ∋ inr l ∣₁ i) = loop (u , v , l) i

    -- and back: a circle is run around the graph, down the tree to the
    -- source of its edge, along the edge, and back up the tree from the
    -- target
    from : Bouquet Loops → ∣ V , E ∣
    from base = ∣ r ∣₀
    from (loop (u , v , l) i) =
      (p u ∙∙ ∣ u ⇒ v ∋ inr l ∣₁ ∙∙ sym (p v)) i

    -- `to` is constant on the tree — pointwise, by its two clauses …
    toConst : (x : ∣ V , T ∣) → to (ι x) ≡ base
    toConst ∣ v ∣₀ = refl
    toConst (∣ u ⇒ v ∋ t ∣₁ i) = refl

    -- … and hence on paths, which is what the section needs: this is the one
    -- place where it matters that `P` runs inside the tree.
    trivial : {v : V} (q : Path (∣ V , T ∣) (∣ r ∣₀) (∣ v ∣₀))
      → Path (Path (Bouquet Loops) base base) (cong to (cong ι q)) refl
    trivial q κ i = toConst (q i) κ

    -- refl ∙∙ q ∙∙ refl is q: the filler of the double composite is a plain
    -- path once its two sides are constant
    ∙∙refl : {x y : Bouquet Loops} (q : x ≡ y) → refl ∙∙ q ∙∙ refl ≡ q
    ∙∙refl q = sym (doubleCompPath-filler refl q refl)

    -- so a circle, run around the graph and contracted again, is itself
    loopEq : (u v : V) (l : L u v)
      → cong to (p u ∙∙ ∣ u ⇒ v ∋ inr l ∣₁ ∙∙ sym (p v))
        ≡ loop (u , v , l)
    loopEq u v l =
        cong-∙∙ to (p u) ∣ u ⇒ v ∋ inr l ∣₁ (sym (p v))
      ∙ (λ κ → trivial (rootPath u) κ
                ∙∙ loop (u , v , l)
                ∙∙ sym (trivial (rootPath v) κ))
      ∙ ∙∙refl (loop (u , v , l))

    sec : (x : Bouquet Loops) → to (from x) ≡ x
    sec base = refl
    sec (loop (u , v , l) i) = λ κ → loopEq u v l κ i

    -- The other direction is the two halves of the collapse: a tree edge is
    -- pushed into the triangle `Fill` gives for it, and an edge of `L` into
    -- the loop that `from` sent its circle to.
    ret : (y : ∣ V , E ∣) → from (to y) ≡ y
    ret ∣ v ∣₀ = p v
    ret (∣ u ⇒ v ∋ inl t ∣₁ i) = λ j → ι (Fill t i j)
    ret (∣ u ⇒ v ∋ inr l ∣₁ i) =
      pqr⁻¹r≡pq (p u) ∣ u ⇒ v ∋ inr l ∣₁ (p v) i
